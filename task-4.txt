package guvitask4;

public class Student {
	static int rollno, age;
	static String name,course;
	Student(int rollno1,int age1,String name1, String course1)
	{
		rollno=rollno1;
		age=age1;
		name=name1;
		course=course1;
	}
		public static void voteAge(int n) throws AgeNotWithinRangeException
		{
			
			if((n<15)&&(n>21))
		
		{
			System.out.println("Age is within the range");		//AgeNotWithinRangeException class created using throw and throws																																										
		}
		else 
		{
			throw new AgeNotWithinRangeException("Age is not within the range");
		}
		
		}
		public static void voteName(String name1) throws NameNotValidException
		{
			name=name1;
			int i,d=0;
			for(i=0;i<name.length();i++)
			{
				char a=name.charAt(i);
								//NameNotValidException-Exception class created using throw and throws
				if(!Character.isLetter(a))
				{
					d++;
				}
			}
			
			if(d==0)
			{
				System.out.println("All are words");
			}else 
			{
				throw new NameNotValidException("The given name is not valid");
			}
		}
				
		
	public static void main(String args[])
	{
		new Student(001,25,"Sangeetha.M","CSE");
	try
	{
		voteAge(25);
	}								//try and catch statements calling exception classes...
	catch(AgeNotWithinRangeException e)
	{
		e.printStackTrace();
	}
	
	try
	{
		voteName("Sangeetha.M");
	}
	catch(NameNotValidException ae)
	{
		ae.printStackTrace();
	}
}
}

package guvitask4;

import java.util.*;

public class HashMap1
	{
	 static Map<String,Integer> h=new HashMap<>();
	static {
	h.put("Sangeetha",10);
	h.put("Geetha",9);
	h.put("Sam",10);
	h.put("Arun",5);						//hashmap...adding the value initially....
	h.put("Rakesh",7);
	}
	
	void add(String name1,int n)
	{								//add method....
		h.put(name1,n);
	}
	void remove(String name1,int n)
	{								//remove method.....
		h.remove(name1,n);
	}
	void display()
	{
		for(Map.Entry<String,Integer> e:h.entrySet())
		{									//display method..
			
			System.out.println(e.getKey()+" "+e.getValue());
		}
	}
	public static void main(String args[])
	{
		HashMap1 hs=new HashMap1();
		System.out.println("....HashMap.....");
		System.out.println("The initial values are:");
		hs.display();
		System.out.println("After adding values:");					//calling methods in main class..
		hs.add("Anitha",10);
		hs.display();
		System.out.println("After removing the specified values:");
		hs.remove("Sam", 10);
		hs.display();
		
	}
	}

package guvitask4;
import java.util.*;

public class Stack1
{
	public static void main(String args[])
	{
		System.out.println("...Stack....");
		boolean result;							//in stack adding valuesss...
		Stack<Integer> st=new Stack<Integer>();
		st.push(78);
		st.push(89);
		st.push(56);
		st.push(80);
		Iterator it=st.iterator();
		while(it.hasNext()) 
		{									//iteration...
			System.out.println(it.next());
		}
		System.out.println("The popped element is"+st.pop());			//pop...
		System.out.println("The peeked element is"+st.peek());				//peek...
		result=st.empty();
		if(result)
		{
			System.out.println("The stack is empty");
		}
		else									//checking the stack is empty.....
		{
			System.out.println("The stack is not empty");
		}
		
	}
}
